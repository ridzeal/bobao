> 🔧 **write_file** — /**
 * GET /api/sessions/[id]/stream
 *
 * Server-Sent Events endpoint.  Each event is a JSON-encoded LogLine:
 *   data: {"ts":"<ISO>","text":"...","level":"info"|"warn"|"error"|"prompt"}\n\n
 *
 * The stream:
 *   1. Immediately replays all buffered lines from session start.
 *   2. Then tails live output until the process exits.
 *   3. Sends a final `event: done\ndata: {}\n\n` when the process ends.
 *
 * The existing LogPanel receives these lines via the client-side
 * useLiveStream hook; no component changes required.
 *
 * Returns 404 when no live process is registered for the given session id.
 */

import { NextRequest } from "next/server";
import { getHandle } from "@/lib/live-bob-provider";
import { LogLine } from "@/lib/session-provider";

export const dynamic = "force-dynamic";

function sseEvent(line: LogLine): string {
  return `data: ${JSON.stringify({ ...line, ts: line.ts.toISOString() })}\n\n`;
}

export async function GET(
  _req: NextRequest,
  { params }: { params: { id: string } },
) {
  const sessionId = params.id;
  const handle = getHandle(sessionId);

  if (!handle) {
    return new Response(
      JSON.stringify({ error: "No live process for session " + sessionId }),
      { status: 404, headers: { "Content-Type": "application/json" } },
    );
  }

  const stream = new ReadableStream({
    async start(controller) {
      function enqueue(line: LogLine) {
        try {
          controller.enqueue(new TextEncoder().encode(sseEvent(line)));
        } catch {
          // Client disconnected
        }
      }

      // 1. Replay buffered lines
      for (const line of handle.lines) {
        enqueue(line);
      }

      // 2. If already done, close immediately
      if (handle.session.status === "done") {
        controller.enqueue(
          new TextEncoder().encode("event: done\ndata: {}\n\n"),
        );
        controller.close();
        return;
      }

      // 3. Subscribe for live lines
      let resolve: ((v: LogLine | null) => void) | null = null;
      const queue: LogLine[] = [];

      function push(line: LogLine) {
        queue.push(line);
        if (resolve) {
          const r = resolve;
          resolve = null;
          r(queue.shift()!);
        }
      }

      handle.subscribers.add(push);

      try {
        while (true) {
          if (queue.length > 0) {
            enqueue(queue.shift()!);
            continue;
          }

          if (handle.session.status === "done") break;

          const next = await new Promise<LogLine | null>((r) => {
            const t = setTimeout(() => {
              if (resolve === r) { resolve = null; r(null); }
            }, 500);
            resolve = (v) => { clearTimeout(t); r(v); };
          });

          if (next !== null) enqueue(next);
          if (handle.session.status === "done" && queue.length === 0) break;
        }
      } finally {
        handle.subscribers.delete(push);
      }

      // Drain any remaining queued lines
      for (const line of queue) {
        enqueue(line);
      }

      controller.enqueue(
        new TextEncoder().encode("event: done\ndata: {}\n\n"),
      );
      controller.close();
    },
  });

  return new Response(stream, {
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      Connection: "keep-alive",
      "X-Accel-Buffering": "no",   // disable nginx buffering if behind a proxy
    },
  });
}

 *

 *
 * Writes `text + "\n"` to the bob process stdin and flips status back to

 *
 * Returns 404 when no live process exists for the session.
 * Returns 409 when the process has already exited.
 */

import { NextRequest, NextResponse } from "next/server";
import { getHandle } from "@/lib/live-bob-provider";

export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } },
) {
  const sessionId = params.id;
  const handle = getHandle(sessionId);

  if (!handle) {
    return NextResponse.json(
      { error: "No live process for session " + sessionId },
      { status: 404 },
    );
  }

  if (handle.session.status === "done") {
    return NextResponse.json(
      { error: "Process has already exited" },
      { status: 409 },
    );
  }

  let body: { text?: unknown };
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 });
  }

  const { text } = body;
  if (typeof text !== "string" || !text.trim()) {
    return NextResponse.json(
      { error: "`text` (non-empty string) is required" },
      { status: 400 },
    );
  }

  handle.proc.stdin.write(text + "\n");
  handle.session.status = "running";

  return NextResponse.json({ ok: true });
}

## What was added

### [`lib/live-bob-provider.ts`](agent-ops/lib/live-bob-provider.ts)

**`LiveBobProvider`** — implements the full `SessionProvider` interface backed by a real child process.

- **`spawnBobProcess(sessionId, args, cwd, meta)`** — spawns `bob` with `stdio: ["pipe","pipe","pipe"]`, wires `readline` on both stdout and stderr, stores the `ProcessHandle` in a module-level `Map<sessionId, ProcessHandle>`. Replaces any previous handle.
- **`pushLine()`** — classifies each line (`info` / `warn` / `error` / `prompt`), buffers it, and fans it out to all active SSE subscribers.
- **Blocked heuristic** — `BLOCKED_PATTERNS: RegExp[]` (configurable array at the top of the file). A line classified as `"prompt"` flips `session.status` to `"blocked"`. Currently checks: lines ending in `?`, `(y/n)`, `waiting for`, `confirm`, `do you want`, `press enter`, `choose …:`, `enter your`.
- **`streamLogs()`** — yields all buffered lines first, then subscribes to the push-queue for live lines; exits cleanly when the process closes.
- **`sendInput()`** — writes `text + "\n"` to `proc.stdin`, flips status back to `"running"`.

### [`lib/composite-provider.ts`](agent-ops/lib/composite-provider.ts)

**`CompositeSessionProvider`** — thin router:

### Usage

```bash
# 1. Spawn bob for the live session
curl -X POST http://localhost:3000/api/sessions/sess-live/spawn \
  -H 'Content-Type: application/json' \
  -d '{"args":["--task","..."], "cwd":"/path/to/repo", "title":"My task"}'

# 2. Stream logs (connect EventSource in the browser or curl)
curl -N http://localhost:3000/api/sessions/sess-live/stream

# 3. Unblock (when banner appears)
curl -X POST http://localhost:3000/api/sessions/sess-live/input \
  -H 'Content-Type: application/json' \
  -d '{"text":"A"}'
```

The SSE stream from `/stream` feeds the existing `LogPanel` — wire it to an `EventSource` on the client side and call `setLines(prev => [...prev, parsed])` whenever a `message` event arrives. The blocked banner surfaces automatically because `getSession()` returns the live `status: "blocked"` from the handle.
